    const semiPerimeter = (SideA + SideB + SideC) / 2;
    const area = Math.sqrt(
    semiPerimeter *
    (semiPerimeter - SideA) *
    (semiPerimeter - SideB) *
    (semiPerimeter - SideC) 
    );
